/** * firebase-curso.js * Lógica compartida para todos los cursos: * - Detecta usuario logueado con Firebase Auth * - Carga y guarda progreso en Firestore * - Controla la navegación entre lecciones * * INSTRUCCIONES: * 1. Reemplaza el objeto firebaseConfig con tus credenciales reales de Firebase. * 2. Asegúrate de haber habilitado Authentication y Firestore en tu proyecto. * 3. Las reglas de seguridad recomendadas están al final de este archivo (comentadas). */ // ====================== CONFIGURACIÓN FIREBASE ====================== // ⚠️ REEMPLAZA ESTO CON TUS CREDENCIALES REALES const firebaseConfig = { apiKey: "TU_API_KEY", authDomain: "TU_PROYECTO.firebaseapp.com", projectId: "TU_PROYECTO_ID", storageBucket: "TU_PROYECTO.appspot.com", messagingSenderId: "TU_SENDER_ID", appId: "TU_APP_ID" }; // Inicializar Firebase (compatibilidad con SDK modular o compat) import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-app.js"; import { getAuth, onAuthStateChanged, signOut } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-auth.js"; import { getFirestore, doc, getDoc, setDoc, updateDoc } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-firestore.js"; const app = initializeApp(firebaseConfig); const auth = getAuth(app); const db = getFirestore(app); // ====================== VARIABLES GLOBALES ====================== let currentUser = null; let courseId = null; // Se define en cada HTML let lessonsData = []; // Se define en cada HTML let currentLessonIndex = 0; let completedLessons = new Set(); // ====================== INICIALIZACIÓN ====================== export function initCourse(id, lessons) { courseId = id; lessonsData = lessons; onAuthStateChanged(auth, async (user) => { if (user) { currentUser = user; document.getElementById("auth-loading")?.classList.add("hidden"); document.getElementById("curso-app")?.classList.remove("hidden"); // Mostrar info del usuario const userNameEl = document.getElementById("user-name"); const userAvatarEl = document.getElementById("user-avatar"); if (userNameEl) userNameEl.textContent = user.displayName || user.email.split("@")[0]; if (userAvatarEl) userAvatarEl.textContent = (user.displayName || user.email)[0].toUpperCase(); // Cargar progreso await loadProgress(); renderSidebar(); showLesson(currentLessonIndex); } else { // No logueado → redirigir o mostrar mensaje document.getElementById("auth-loading")?.classList.add("hidden"); document.getElementById("no-auth")?.classList.remove("hidden"); } }); } // ====================== PROGRESO ====================== async function loadProgress() { if (!currentUser || !courseId) return; try { const progressRef = doc(db, "users", currentUser.uid, "progress", courseId); const snap = await getDoc(progressRef); if (snap.exists()) { const data = snap.data(); completedLessons = new Set(data.completedLessons || []); currentLessonIndex = data.lastLessonIndex ?? 0; } else { // Crear documento inicial await setDoc(progressRef, { completedLessons: [], lastLessonIndex: 0, percentage: 0, updatedAt: new Date().toISOString() }); } } catch (err) { console.error("Error cargando progreso:", err); } } async function saveProgress() { if (!currentUser || !courseId) return; const percentage = Math.round((completedLessons.size / lessonsData.length) * 100); try { const progressRef = doc(db, "users", currentUser.uid, "progress", courseId); await setDoc(progressRef, { completedLessons: Array.from(completedLessons), lastLessonIndex: currentLessonIndex, percentage, updatedAt: new Date().toISOString() }, { merge: true }); } catch (err) { console.error("Error guardando progreso:", err); } } // ====================== RENDERIZADO ====================== function renderSidebar() { const container = document.getElementById("modulos-lista"); if (!container) return; // Agrupar por módulo const modulos = {}; lessonsData.forEach((lesson, index) => { if (!modulos[lesson.modulo]) modulos[lesson.modulo] = []; modulos[lesson.modulo].push({ ...lesson, index }); }); let html = ""; Object.entries(modulos).forEach(([nombreModulo, lecciones]) => { html += `
${nombreModulo}
`; lecciones.forEach(l => { const completada = completedLessons.has(l.index); const activa = l.index === currentLessonIndex; html += `
${completada ? '✓' : ''}
${l.index + 1} ${l.titulo}
`; }); html += `
`; }); container.innerHTML = html; // Actualizar barra de progreso const percent = Math.round((completedLessons.size / lessonsData.length) * 100); const fill = document.getElementById("progress-fill"); const percentText = document.getElementById("progress-percent"); if (fill) fill.style.width = percent + "%"; if (percentText) percentText.textContent = percent + "%"; } function showLesson(index) { if (index < 0 || index >= lessonsData.length) return; currentLessonIndex = index; const lesson = lessonsData[index]; // Actualizar título y breadcrumb document.getElementById("leccion-titulo").textContent = lesson.titulo; document.getElementById("breadcrumb").textContent = `${lesson.modulo} → Lección ${index + 1}`; // Contenido document.getElementById("leccion-contenido").innerHTML = lesson.contenido; // Botones const btnPrev = document.getElementById("btn-prev"); const btnNext = document.getElementById("btn-next"); const btnComplete = document.getElementById("btn-complete"); btnPrev.disabled = index === 0; btnPrev.style.opacity = index === 0 ? "0.5" : "1"; if (index === lessonsData.length - 1) { btnNext.textContent = "Finalizar curso"; } else { btnNext.textContent = "Siguiente lección →"; } // Estado del botón completar if (completedLessons.has(index)) { btnComplete.textContent = "✓ Completada"; btnComplete.classList.add("completada"); } else { btnComplete.textContent = "Marcar como completada"; btnComplete.classList.remove("completada"); } renderSidebar(); saveProgress(); // Guardar última posición window.scrollTo({ top: 0, behavior: "smooth" }); } // ====================== ACCIONES GLOBALES ====================== window.goToLesson = function(index) { showLesson(index); }; window.prevLesson = function() { if (currentLessonIndex > 0) showLesson(currentLessonIndex - 1); }; window.nextLesson = function() { // Marcar actual como completada al avanzar if (!completedLessons.has(currentLessonIndex)) { completedLessons.add(currentLessonIndex); } if (currentLessonIndex < lessonsData.length - 1) { showLesson(currentLessonIndex + 1); } else { // Última lección saveProgress(); alert("¡Felicidades! Has completado el curso 🎉\nTu progreso está guardado."); } }; window.markComplete = function() { if (completedLessons.has(currentLessonIndex)) { completedLessons.delete(currentLessonIndex); } else { completedLessons.add(currentLessonIndex); } renderSidebar(); saveProgress(); showLesson(currentLessonIndex); // refrescar botón }; window.logout = async function() { await signOut(auth); window.location.href = "/"; // o a tu página de login }; // ====================== REGLAS DE SEGURIDAD RECOMENDADAS (Firestore) ====================== /* rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { match /users/{userId}/progress/{courseId} { allow read, write: if request.auth != null && request.auth.uid == userId; } } } */
